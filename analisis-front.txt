/app
    vistas
    (drawer para diferentes ventanas)
        HomeScreen - Muestra recetas guardadas
        RecipeDetailScreen - Detalle de una receta
        GenerateRecipeScreen - Formulario para generar receta con IA
        
        ProfileScreen - Datos del usuario

/core
    TYPES.ts - Símbolos únicos para cada binding de Inversify
        IItemRepository: Symbol
        IUserRepository: Symbol
        IRecipeRepository: Symbol
        ItemUseCase: Symbol
        UserUseCase: Symbol
        RecipeUseCase: Symbol
        
    container.ts - Vincula interfaces con implementaciones concretas
        bind<IItemRepository> -> ItemRepository
        bind<IUserRepository> -> UserRepository
        bind<IRecipeRepository> -> RecipeRepository
        

        bind<ItemUseCase> inject(IItemRepository)
        bind<RecipeUseCase> inject(IRecipeRepository, IAIService)

/data
    /database
        connection.ts
            +string getConnection() - Retorna la instancia/string de conexión a la DB

    /repositories
        ItemRepository.ts implements IItemRepository
            +Item[] get()
            +Item getById(id: string)
            +void post(item: Item)
            +void update(item: Item)
            +void delete(id: string)
        UserRepository.ts implements IUserRepository
            +User get()
            +void post(user: User)
        RecipeRepository.ts implements IRecipeRepository
            +Recipe[] get()
            +void post(request : RecipeCreateRequestDTO)
            +void delete(id: string)
