/app
    vistas
    (drawer para diferentes ventanas)
        HomeScreen - Muestra recetas guardadas
        RecipeDetailScreen - Detalle de una receta
        GenerateRecipeScreen - Formulario para generar receta con IA
        InventoryScreen - Gestión del inventario
        ProfileScreen - Datos del usuario

/core
    TYPES.ts - Símbolos únicos para cada binding de Inversify
        IItemRepository: Symbol
        IUserRepository: Symbol
        IRecipeRepository: Symbol
        IInventoryRepository: Symbol
        IAIApi: Symbol
        ItemUseCase: Symbol
        UserUseCase: Symbol
        RecipeUseCase: Symbol
        InventoryUseCase: Symbol
    container.ts - Vincula interfaces con implementaciones concretas
        bind<IItemRepository> -> ItemRepository
        bind<IUserRepository> -> UserRepository
        bind<IRecipeRepository> -> RecipeRepository
        bind<IInventoryRepository> -> InventoryRepository
        bind<IAIApi> -> AIapi
        bind<ItemUseCase> inject(IItemRepository)
        bind<RecipeUseCase> inject(IRecipeRepository, IAIApi)

/data
    /database
        connection.ts
            +string getConnection() - Retorna la instancia/string de conexión a la DB

    /AI
        API_Keys.ts - Claves de API del proveedor de IA (no subir a git)
            OPENAI_KEY: string
            MODEL: string
        AIapi.ts implements IAIApi
            +void prompt(text: string) - Envía el texto al modelo de IA
            +Recipe response() - Parsea y retorna la respuesta como entidad Recipe

    /repositories
        ItemRepository.ts implements IItemRepository
            +Item[] get()
            +Item getById(id: string)
            +void post(item: Item)
            +void update(item: Item)
            +void delete(id: string)
        UserRepository.ts implements IUserRepository
            +User get()
            +void post(user: User)
        RecipeRepository.ts implements IRecipeRepository
            +Recipe[] get()
            +void post(recipe: Recipe)
            +void delete(id: string)


/domain
    /dto
    /entities
        user.ts
            -id: string
            -name: string
            -email: string
        recipe.ts
            -id: string
            -name: string
            -ingredients: string[]
            -steps: string[]
            -userId: string
        item.ts
            -id: string
            -name: string


    /usecases
        ItemUseCase.ts implements IItemUseCase inject(IItemRepository)
            +Item[] get()
            +void post(item: Item)
            +void update(item: Item)
            +void delete(id: string)
        UserUseCase.ts implements IUserUseCase inject(IUserRepository)
            +User get()
            +void post(user: User)
        RecipeUseCase.ts implements IRecipeUseCase inject(IRecipeRepository, IAIApi)
            +Recipe[] get()
            +Recipe post(prompt: string) - Llama a IAIApi, guarda resultado y lo retorna
            +void delete(id: string)
