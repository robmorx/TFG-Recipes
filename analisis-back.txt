        /repository (repositorios por tabla en base de datos)
        (Implementaciones concretas de las interfaces del dominio. Acceso real a la DB vía Sequelize.)

            user.repository.ts implements IUserRepository @Injectable()
                constructor inject(@InjectModel(UserModel))
                +User[] getList() - Consulta findAll() a la tabla users
                +User getById(id: number) - findOne() por user_id
                +User getByUUID(uuid: UUID) - findOne() por user_uuid
                +number add(entity: User) - create() + save() en la tabla users
                +number delete(id: number) - findOne() + destroy()
                +number update(id: number, entity: User) - update() donde user_id = id

            recipe.repository.ts implements IRecipeRepository @Injectable()
                constructor inject(@InjectModel(RecipeModel))
                +Recipe[] getList() - findAll() de todas las recetas
                +Recipe getByUUID(uuid: UUID) - findOne() por recipe_uuid
                +Recipe[] getByUserUUID(user_uuid: UUID) - findAll() filtrado por user_uuid
                +number add(entity: Recipe) - create() + save()
                +number delete(id: number) - findOne() + destroy()

            item.repository.ts implements IItemRepository @Injectable()
                constructor inject(@InjectModel(ItemModel))
                +Item[] getList() - findAll()
                +Item getByUUID(uuid: UUID) - findOne() por item_uuid
                +number add(entity: Item) - create() + save()
                +number delete(id: number) - findOne() + destroy()
                +number update(id: number, entity: Item) - update() donde item_id = id

                    recipe.controller.ts @Controller('api/recipes')
                        @Get getList() - GET /api/recipes

                        @Delete delete(@Body RecipeDeleteRequestDTO) - DELETE /api/recipes/delete
