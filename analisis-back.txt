/src
    /data
        /AI
            (Servicio externo de IA, consumido por RecipeUseCase)
            ai.service.ts @Injectable()
                constructor inject(API_KEY, MODEL)
                +Recipe generate(prompt: string) - Llama a la API de IA y parsea respuesta como entidad Recipe
            ai.config.ts - Centraliza API_KEY y MODEL del proveedor (no subir a git, usar .env)

        /database (conexion a base de datos, NO HACE FALTA POR EL .env)

        /repository (repositorios por tabla en base de datos)
        (Implementaciones concretas de las interfaces del dominio. Acceso real a la DB vía Prisma.)

            user.repository.ts implements IUserRepository @Injectable()
                constructor inject(PrismaService)
                +User[] getList() - findMany() en la tabla users
                +User getById(id: number) - findUnique() por user_id
                +User getByUUID(uuid: UUID) - findUnique() por user_uuid
                +number add(entity: User) - create() en la tabla users
                +number delete(id: number) - findUnique() + delete()
                +number update(id: number, entity: User) - update() donde user_id = id

            recipe.repository.ts implements IRecipeRepository @Injectable()
                constructor inject(PrismaService)
                +Recipe[] getList() - findMany() de todas las recetas
                +Recipe getByUUID(uuid: UUID) - findUnique() por recipe_uuid
                +Recipe[] getByUserUUID(user_uuid: UUID) - findMany() filtrado por user_uuid
                +number add(entity: Recipe) - create()
                +number delete(id: number) - findUnique() + delete()

            inventory.repository.ts implements IInventoryRepository @Injectable()
                constructor inject(PrismaService)
                +Inventory getByUserUUID(user_uuid: UUID) - findUnique() por user_uuid con include items
                +number add(entity: Inventory) - create() en la tabla inventory (se crea al registrar usuario)
                +number delete(id: number) - delete() del inventario y sus items en cascada

            item.repository.ts implements IItemRepository @Injectable()
                constructor inject(PrismaService)
                +Item getById(id: number) - findUnique() por item_id
                +number add(entity: Item) - create() en la tabla items con FK inventory_id
                +number delete(id: number) - findUnique() + delete()
                +number update(id: number, entity: Item) - update() nombre y/o cantidad donde item_id = id


    /domain

        /user
            user.ts (entity)
                -id: number
                -user_uuid: UUID
                -name: string
                -email: string
                -password: string
                -atcreated: Date
                -atmodified: Date
            /dto
                user.add.request.dto.ts - Valida datos de entrada para crear usuario

                    @IsString @IsNotEmpty @IsEmail email: string
                    @IsString @IsNotEmpty password: string
                user.update.request.dto.ts - Valida datos de entrada para actualizar usuario
                    @IsUUID user_uuid: UUID
                    @IsString name: string
                    @IsString @IsEmail email: string
                user.delete.request.dto.ts
                    @IsUUID user_uuid: UUID
                user.getbyuuid.request.dto.ts
                    @IsUUID user_uuid: UUID
                user.response.dto.ts - Formato de salida al cliente (nunca expone password ni id interno)
                    +user_uuid: UUID
                    +name: string
                    +email: string
                    +atcreated: Date
            /interfaces
                iuser.usecase.ts - Contrato para UseCase y Controller
                    +UserResponseDTO[] getList()
                    +UserResponseDTO getByUUID(uuid: UUID)
                    +number add(entity: UserAddRequestDTO)
                    +number delete(uuid: UUID)
                    +number update(entity: UserUpdateRequestDTO)
                iuser.repository.ts - Contrato para la capa data
                    +User[] getList()
                    +User getById(id: number)
                    +User getByUUID(uuid: UUID)
                    +number add(entity: User)
                    +number delete(id: number)
                    +number update(id: number, entity: User)
            user.usecases.ts implements IUserUseCase @Injectable()
                constructor inject(UserRepository)
                +UserResponseDTO[] getList() - Obtiene lista y mapea a ResponseDTO
                +UserResponseDTO getByUUID(uuid) - Busca por UUID y mapea a ResponseDTO
                +number add(entity) - Construye entidad User con nuevo UUID y delega al repo
                +number delete(uuid) - Busca UUID, obtiene id interno y delega al repo
                +number update(entity) - Busca UUID, construye entidad actualizada y delega

        /inventory
            inventory.ts (entity)
                -id: number
                -inventory_uuid: UUID

                    +user_uuid: UUID
                    +items: ItemResponseDTO[]
                    +atcreated: Date
            /interfaces
                iinventory.usecase.ts
                    +InventoryResponseDTO getByUserUUID(user_uuid: UUID)
                    +number add(user_uuid: UUID) - Crea inventario vacío al registrar usuario
                    +number delete(uuid: UUID) - Elimina inventario y sus items en cascada
                iinventory.repository.ts - Contrato para la capa data
                    +Inventory getByUserUUID(user_uuid: UUID)
                    +number add(entity: Inventory)
                    +number delete(id: number)
            inventory.usecases.ts implements IInventoryUseCase @Injectable()
                constructor inject(InventoryRepository)
                +InventoryResponseDTO getByUserUUID(user_uuid) - Recupera inventario con items y mapea
                +number add(user_uuid) - Construye entidad Inventory con nuevo UUID y delega al repo
                +number delete(uuid) - Busca UUID, obtiene id interno y elimina en cascada

        /item
            item.ts (entity)
                -id: number              (PK interna, usada para update y delete)
                -inventory_id: number    (FK -> inventory, nunca expuesta al exterior)
                -name: string
                -quantity: number
                -atcreated: Date
                -atmodified: Date
            /dto
                item.add.request.dto.ts
                    @IsUUID inventory_uuid: UUID  (el front la obtiene del inventario del usuario)
                    @IsString @IsNotEmpty name: string
                    @IsNumber @Min(1) quantity: number
                item.update.request.dto.ts
                    @IsNumber id: number
                    @IsString name: string
                    @IsNumber @Min(0) quantity: number
                item.delete.request.dto.ts
                    @IsNumber id: number
                item.response.dto.ts - Se devuelve anidado dentro de InventoryResponseDTO
                    +id: number
                    +name: string
                    +quantity: number
            /interfaces
                iitem.usecase.ts
                    +number add(entity: ItemAddRequestDTO)
                    +number delete(id: number)
                    +number update(entity: ItemUpdateRequestDTO)
                iitem.repository.ts - Contrato para la capa data
                    +Item getById(id: number)
                    +number add(entity: Item)
                    +number delete(id: number)
                    +number update(id: number, entity: Item)
            item.usecases.ts implements IItemUseCase @Injectable()
                constructor inject(ItemRepository, InventoryRepository)
                +number add(entity) - Resuelve inventory_id a partir de inventory_uuid y crea el item
                +number delete(id) - Elimina el item por id interno
                +number update(entity) - Actualiza nombre y/o cantidad del item


    /presentation
        /controllers
            recipe.controller.ts @Controller('api/recipes')
                @Get getList() - GET /api/recipes
                @Get getByUser(:user_uuid) - GET /api/recipes/:user_uuid
                @Post add(@Body RecipeAddRequestDTO) - POST /api/recipes/add
                @Delete delete(@Body RecipeDeleteRequestDTO) - DELETE /api/recipes/delete

            inventory.controller.ts @Controller('api/inventory')
                @Get getByUser(:user_uuid) - GET /api/inventory/:user_uuid → devuelve inventario con items anidados
                @Delete delete(@Body) - DELETE /api/inventory/delete → elimina inventario y sus items en cascada

            item.controller.ts @Controller('api/items')
                @Post add(@Body ItemAddRequestDTO) - POST /api/items/add
                @Delete delete(@Body ItemDeleteRequestDTO) - DELETE /api/items/delete
                @Put update(@Body ItemUpdateRequestDTO) - PUT /api/items/update

            user.controller.ts @Controller('api/users')
                @Get getList() - GET /api/users
                @Get getUser(:user_uuid) - GET /api/users/:user_uuid
                @Post add(@Body UserAddRequestDTO) - POST /api/users/add
                @Delete delete(@Body UserDeleteRequestDTO) - DELETE /api/users/delete
                @Put update(@Body UserUpdateRequestDTO) - PUT /api/users/update
